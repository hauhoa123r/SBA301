"""Read-only Compose checks and rollout/rollback failure injection.

No engine, network or production credentials are used. Compose config tests use
the real CLI; deployment tests execute the real transaction against a fake engine.
"""
import copy
from contextlib import redirect_stdout, redirect_stderr
import hashlib
import importlib.util
import io
import json
import os
from pathlib import Path
import shutil
import subprocess
import tempfile
import unittest
from unittest.mock import patch

ROOT = Path(__file__).resolve().parents[2]
module = importlib.util.spec_from_file_location("release", ROOT / "deployment/release.py")
release = importlib.util.module_from_spec(module)
module.loader.exec_module(release)


def digest(value):
    return "sha256:" + hashlib.sha256(value.encode()).hexdigest()


def image(service, tag):
    spec = {"tag": tag, "id": digest(service + tag),
            "ref": release.REPOS[service] + "@" + digest("manifest-" + service + tag)}
    if service == "nginx":
        spec["config_sha256"] = hashlib.sha256(("nginx" + tag).encode()).hexdigest()
    return spec


def version(tag="21"):
    return {"image_tag": tag, "images": {name: image(name, tag) for name in release.REPOS},
            "backend_ddl_mode": "none" if tag == "21" else "validate"}


class FakeEngine:
    def __init__(self, root, payload):
        self.root, self.payload = root, payload
        self.calls = []
        self.up_calls = []
        self.fail = None
        self.items = {}
        self.images = {}
        for tag in ("21", "22"):
            for service in release.REPOS:
                spec = image(service, tag)
                info = {"Id": spec["id"], "RepoDigests": [spec["ref"]], "spec": spec}
                for ref in (spec["ref"], spec["id"], release.REPOS[service] + ":" + tag):
                    self.images[ref] = info
        for service in (*release.REPOS, "prometheus", "grafana"):
            spec = image(service, "21") if service in release.REPOS else {"id": digest(service)}
            self.items[service] = {
                "Id": digest("container-" + service + "21"), "Image": spec["id"],
                "Config": {"Image": (release.REPOS.get(service, service) + ":21"),
                           "Labels": {"com.docker.compose.project": "sba301",
                                      "com.docker.compose.service": service,
                                      "com.docker.compose.project.config_files": str(root / "docker-compose.yml")}},
                "State": {"Status": "running", "Health": {"Status": "healthy"}},
                "Mounts": [{"Type": "volume", "Name": "sba301_" + service + "_data",
                            "Destination": "/var/lib/" + service}],
            }

    def config(self, env):
        services = {service: {"image": env[service.upper() + "_IMAGE_REF"], "volumes": [service + "-volume"]}
                    for service in release.REPOS}
        services.update(prometheus={"image": "prom/prometheus:v3.0.0", "volumes": ["prometheus_data"]},
                        grafana={"image": "grafana/grafana:11.0.0", "volumes": ["grafana_data"]})
        if self.fail == "compose-image":
            services["backend"]["image"] = "devhoa/chinese-backend:old"
        if self.fail == "app-mount" and env["BACKEND_IMAGE_TAG"] == "22":
            services["backend"]["volumes"] = ["unexpected-new-volume"]
        return {"services": services, "volumes": {"mysql_data": {}, "prometheus_data": {}, "grafana_data": {}},
                "networks": {"chinese-net": {}}}

    def command(self, args, *, env=None, data=None, cwd=None):
        args = [str(a) for a in args]
        self.calls.append(args)
        if args[:2] == ["docker", "inspect"]:
            values = []
            for key in args[2:]:
                if key.startswith("chinese-"):
                    values.append(self.items[key.removeprefix("chinese-")])
                else:
                    values.append(next(item for item in self.items.values() if item["Id"] == key))
            return json.dumps(values)
        if args[:3] == ["docker", "image", "inspect"]:
            info = self.images[args[3]]
            if self.fail == "pulled-image" and info["spec"]["tag"] == "22":
                return json.dumps([dict(info, Id=digest("wrong"))])
            return json.dumps([info])
        if args[:2] == ["docker", "ps"]:
            return "\n".join(item["Id"] for item in self.items.values())
        if args[:2] == ["docker", "exec"]:
            if "sha256sum" in args:
                return image("nginx", "21")["config_sha256"] + "  /etc/nginx/conf.d/default.conf"
            self.schema_sql = data
            return "column mismatch: lesson_progress.position_seconds" if self.fail == "schema" else ""
        if args[:2] == ["docker", "compose"]:
            if "config" in args:
                return json.dumps(self.config(env))
            if "run" in args and "sha256sum" in args:
                checksum = self.images[env["NGINX_IMAGE_REF"]]["spec"]["config_sha256"]
                return ("0" * 64 if self.fail == "config-mount" else checksum) + "  /etc/nginx/conf.d/default.conf"
            if "up" in args:
                service = args[-1]
                spec = self.images[env[service.upper() + "_IMAGE_REF"]]["spec"]
                self.up_calls.append((service, spec["tag"]))
                item = self.items[service]
                item["Image"] = spec["id"]
                item["Id"] = digest("container-" + service + spec["tag"])
                item["Config"]["Image"] = spec["ref"]
                item["Config"]["Labels"]["com.docker.compose.project.config_files"] = ",".join(
                    [str(self.root / "docker-compose.yml"), str(self.payload / "compose.production.yml")])
                if self.fail == "health" and spec["tag"] == "22":
                    raise release.ReleaseError("injected health timeout")
                if self.fail == "running-image" and spec["tag"] == "22":
                    item["Image"] = digest("wrong-running-image")
                if self.fail == "running-ref" and spec["tag"] == "22":
                    item["Config"]["Image"] = release.REPOS[service] + ":21"
            return ""
        if args[0] == "sh":
            if self.fail == "smoke" and self.items["nginx"]["Image"] == image("nginx", "22")["id"]:
                raise release.ReleaseError("injected TLS smoke failure")
            return "HTTPS verified"
        if args[0] == "curl":
            if args[-1].endswith("/health"):
                return '{"status":"UP"}'
            if args[-1].endswith("/plans"):
                return '[]'
            return '<html>React SPA</html>'
        raise AssertionError("Unexpected command: " + repr(args))


class TransactionTest(unittest.TestCase):
    def setUp(self):
        self.temporary = tempfile.TemporaryDirectory(prefix="release-test-")
        self.addCleanup(self.temporary.cleanup)
        self.root = Path(self.temporary.name)
        self.payload = self.root / "deployment/releases/22"
        self.payload.mkdir(parents=True)
        (self.root / ".env").write_text("IMAGE_TAG=19\nMYSQL_IMAGE_TAG=18\nSECRET=unchanged\n")
        (self.root / "docker-compose.yml").write_text("services: {}\n")
        for name in ("compose.production.yml", "verify-nginx-https.sh"):
            shutil.copyfile(ROOT / "deployment" / name, self.payload / name)
        shutil.copyfile(ROOT / "database/tests/check-schema.sql", self.payload / "check-schema.sql")
        self.secret_bytes = (self.root / ".env").read_bytes()
        self.engine = FakeEngine(self.root, self.payload)
        self.patch = patch.object(release, "command", side_effect=self.engine.command)
        self.patch.start()
        self.addCleanup(lambda: self.patch.stop())
        self.output = io.StringIO()
        self.stdout = redirect_stdout(self.output)
        self.stderr = redirect_stderr(self.output)
        self.stdout.__enter__()
        self.stderr.__enter__()
        self.addCleanup(self.stdout.__exit__, None, None, None)
        self.addCleanup(self.stderr.__exit__, None, None, None)
        self.deployer = release.Deployer(self.root, self.payload)
        self.manifest = version("22")

    def assert_protected_unchanged(self, initial):
        for service in ("mysql", "prometheus", "grafana"):
            self.assertEqual(initial[service], self.engine.items[service])
        self.assertEqual(self.secret_bytes, (self.root / ".env").read_bytes())
        self.assertNotIn("SECRET=", self.output.getvalue())
        self.assertFalse(any("down" in call or "prune" in call or "-v" in call for call in self.engine.calls))

    def test_app_22_keeps_mysql_21_even_when_manifest_contains_mysql_22(self):
        initial = copy.deepcopy(self.engine.items)
        self.deployer.deploy_application(self.manifest)
        stored = release.read_json(self.deployer.version_path)
        self.assertEqual("22", stored["images"]["backend"]["tag"])
        self.assertEqual("22", stored["images"]["nginx"]["tag"])
        self.assertEqual("21", stored["images"]["mysql"]["tag"])
        self.assertEqual("validate", stored["backend_ddl_mode"])
        self.assertEqual([("backend", "22"), ("nginx", "22")], self.engine.up_calls)
        self.assert_protected_unchanged(initial)
        # A new process detects the base files, ignoring the previous managed overlay.
        restarted = release.Deployer(self.root, self.payload)
        self.assertEqual(stored, restarted.snapshot())

    def test_nginx_only_keeps_backend_and_mysql_21(self):
        initial = copy.deepcopy(self.engine.items)
        self.deployer.deploy_application(self.manifest, nginx_only=True)
        stored = release.read_json(self.deployer.version_path)
        self.assertEqual("21", stored["images"]["backend"]["tag"])
        self.assertEqual(initial["backend"], self.engine.items["backend"])
        self.assertEqual([("nginx", "22")], self.engine.up_calls)
        self.assert_protected_unchanged(initial)

    def test_failures_after_rollout_restore_images_and_version(self):
        for failure in ("health", "running-image", "running-ref", "smoke"):
            with self.subTest(failure=failure):
                # Restore the disposable fixture; never touches a real engine.
                self.engine = FakeEngine(self.root, self.payload)
                self.patch.stop()
                self.patch = patch.object(release, "command", side_effect=self.engine.command)
                self.patch.start()
                if self.deployer.version_path.exists():
                    self.deployer.version_path.unlink()
                self.deployer = release.Deployer(self.root, self.payload)
                initial = copy.deepcopy(self.engine.items)
                self.engine.fail = failure
                with self.assertRaises(release.ReleaseError):
                    self.deployer.deploy_application(self.manifest)
                stored = release.read_json(self.deployer.version_path)
                self.assertEqual("none", stored["backend_ddl_mode"])
                for service in ("backend", "nginx"):
                    self.assertEqual(image(service, "21")["id"], self.engine.items[service]["Image"])
                    self.assertEqual("21", stored["images"][service]["tag"])
                self.assert_protected_unchanged(initial)

    def test_preflight_failures_do_not_recreate_any_container(self):
        for failure in ("schema", "pulled-image", "config-mount", "compose-image", "app-mount"):
            with self.subTest(failure=failure):
                self.engine.fail = failure
                with self.assertRaises(release.ReleaseError):
                    self.deployer.deploy_application(self.manifest)
                self.assertEqual([], self.engine.up_calls)
                self.assertEqual(self.secret_bytes, (self.root / ".env").read_bytes())

    def test_existing_version_drift_blocks_deployment(self):
        stored = version()
        stored["images"]["mysql"] = image("mysql", "22")
        release.atomic_json(self.deployer.version_path, stored)
        with self.assertRaisesRegex(release.ReleaseError, "Running mysql differs"):
            self.deployer.deploy_application(self.manifest)
        self.assertEqual([], self.engine.up_calls)

    def test_database_upgrade_requires_approval(self):
        for approval in ({}, {"release_tag": "22", "approved_by": "", "backup_reference": "s3://backup/key"}):
            with self.assertRaises((release.ReleaseError, KeyError)):
                self.deployer.deploy_database(self.manifest, approval)
        self.assertEqual([], self.engine.up_calls)

    def test_approved_database_upgrade_preserves_volumes_and_apps(self):
        initial = copy.deepcopy(self.engine.items)
        self.deployer.deploy_database(self.manifest, {"release_tag": "22", "approved_by": "db-admin",
                                                     "backup_reference": "s3://backups/verified"})
        stored = release.read_json(self.deployer.version_path)
        self.assertEqual("22", stored["images"]["mysql"]["tag"])
        self.assertEqual("none", stored["backend_ddl_mode"])
        self.assertEqual("21", stored["images"]["backend"]["tag"])
        self.assertEqual(initial["mysql"]["Mounts"], self.engine.items["mysql"]["Mounts"])
        for service in ("backend", "nginx", "prometheus", "grafana"):
            self.assertEqual(initial[service], self.engine.items[service])
        self.assertEqual([("mysql", "22")], self.engine.up_calls)
        self.assertEqual(self.secret_bytes, (self.root / ".env").read_bytes())

    def test_latest_and_unpinned_refs_are_rejected(self):
        for tag in ("latest", "LATEST", "", "22;rm", "$(echo unsafe)"):
            with self.assertRaises(release.ReleaseError):
                release.valid_tag(tag)
        bad = image("backend", "22")
        bad["ref"] = "devhoa/chinese-backend:22"
        with self.assertRaises(release.ReleaseError):
            release.validate_spec("backend", bad)
        for mode in ("update", "create", "create-drop"):
            unsafe = version("22")
            unsafe["backend_ddl_mode"] = mode
            with self.assertRaises(release.ReleaseError):
                release.compose_environment(unsafe)

    def test_prepare_release_refuses_tag_moved_after_build(self):
        record = self.root / "built.json"
        document = {"image_tag": "22", "images": {"backend": image("backend", "21")}}
        document["images"]["backend"]["tag"] = "22"
        release.atomic_json(record, document)
        with self.assertRaisesRegex(release.ReleaseError, "changed since build"):
            release.prepare_release(record, self.root / "release.json")

    def test_manual_rollback_keeps_database_upgraded_separately(self):
        self.deployer.deploy_application(self.manifest)
        upgraded = release.Deployer(self.root, self.payload)
        upgraded.deploy_database(self.manifest, {"release_tag": "22", "approved_by": "db-admin",
                                               "backup_reference": "s3://backups/verified"})
        deployer = release.Deployer(self.root, self.payload)
        deployer.rollback("22")
        stored = release.read_json(deployer.version_path)
        self.assertEqual("21", stored["images"]["backend"]["tag"])
        self.assertEqual("21", stored["images"]["nginx"]["tag"])
        self.assertEqual("22", stored["images"]["mysql"]["tag"])

    def test_rollback_of_another_release_is_rejected(self):
        self.deployer.deploy_application(self.manifest)
        deployer = release.Deployer(self.root, self.payload)
        before = list(self.engine.up_calls)
        with self.assertRaisesRegex(release.ReleaseError, "different release"):
            deployer.rollback("23")
        self.assertEqual(before, self.engine.up_calls)


@unittest.skipUnless(shutil.which("docker"), "Docker Compose CLI is required; no daemon needed")
class RealComposeTest(unittest.TestCase):
    def test_real_compose_app_22_mysql_21_overrides_stale_secret_env(self):
        with tempfile.TemporaryDirectory(prefix="compose-versions-") as directory:
            directory = Path(directory)
            secret_env = directory / "secrets.env"
            required = set(__import__("re").findall(r"\$\{([A-Z0-9_]+):\?", (ROOT / "docker-compose.yml").read_text()))
            values = {name: "fixture" for name in required}
            values.update(IMAGE_TAG="19", MYSQL_IMAGE_TAG="18", BACKEND_IMAGE_TAG="17", NGINX_IMAGE_TAG="16")
            secret_env.write_text("\n".join(f"{key}={value}" for key, value in values.items()))
            before = secret_env.read_bytes()
            selected = version("22")
            selected["images"]["mysql"] = image("mysql", "21")
            env = release.compose_environment(selected)
            env["DOCKER_CONFIG"] = str(directory)
            # Desktop can install Compose outside global plugin directories.
            # Use a fresh, non-secret Docker config while preserving discovery.
            plugin_dir = Path(shutil.which("docker")).resolve().parent.parent / "cli-plugins"
            if plugin_dir.is_dir():
                (directory / "config.json").write_text(json.dumps({"cliPluginsExtraDirs": [str(plugin_dir)]}))
            monitoring = directory / "monitoring.yml"
            monitoring.write_text('''services:
  prometheus:
    image: prom/prometheus:v3.0.0
    volumes: [prometheus_data:/prometheus]
    networks: [chinese-net]
  grafana:
    image: grafana/grafana:11.0.0
    volumes: [grafana_data:/var/lib/grafana]
    networks: [chinese-net]
volumes:
  prometheus_data: {}
  grafana_data: {}
''')
            base_command = ["docker", "compose", "--env-file", str(secret_env), "-f", str(ROOT / "docker-compose.yml"),
                            "-f", str(monitoring)]
            result = subprocess.run(base_command + ["config", "--format", "json"],
                                    text=True, capture_output=True, env=env)
            self.assertEqual(0, result.returncode, result.stderr)
            base = json.loads(result.stdout)
            for service in release.REPOS:
                self.assertEqual(release.REPOS[service] + ":" + selected["images"][service]["tag"],
                                 base["services"][service]["image"])
            command = base_command + ["-f", str(ROOT / "deployment/compose.production.yml"), "config", "--format", "json"]
            result = subprocess.run(command, text=True, capture_output=True, env=env)
            self.assertEqual(0, result.returncode, result.stderr)
            config = json.loads(result.stdout)
            for service in release.REPOS:
                self.assertEqual(selected["images"][service]["ref"], config["services"][service]["image"])
                self.assertEqual(base["services"][service].get("volumes"), config["services"][service].get("volumes"))
            for service in ("grafana", "prometheus"):
                self.assertEqual(base["services"][service], config["services"][service])
            self.assertEqual(base["volumes"], config["volumes"])
            self.assertEqual(base["networks"], config["networks"])
            self.assertEqual("22", env["BACKEND_IMAGE_TAG"])
            self.assertEqual("21", env["MYSQL_IMAGE_TAG"])
            self.assertEqual(before, secret_env.read_bytes())
            self.assertIn("mysql_data", config["volumes"])
            self.assertEqual("validate", config["services"]["backend"]["environment"]["SPRING_JPA_HIBERNATE_DDL_AUTO"])
            self.assertIn("healthcheck", config["services"]["backend"])
            self.assertIn("healthcheck", config["services"]["nginx"])
